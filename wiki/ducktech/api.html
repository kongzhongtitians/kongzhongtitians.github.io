API

粉碎机

    {

    "type": "ducktech:shredder",

    "input": [

    {

    "item": "输入物品A"

    },

    {

    "item": "输入物品B"

    }

    ],

    "output": [

    {

    "item": "输出物品",

    "count": 输出数量

    }

    ]

    }

高级粉碎机

    "input": [

    {

    "item": "输入物品A"

    },

    {

    "item": "输入物品B"

    }

    ],

    "output": [

    {

    "item": "输出物品",

    "count": 输出数量

    }

    ],

    "processingTime" : 工作时长，单位为tick

    }
